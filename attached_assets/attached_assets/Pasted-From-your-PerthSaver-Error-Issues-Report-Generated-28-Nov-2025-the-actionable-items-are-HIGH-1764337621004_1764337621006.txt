// client/src/components/Avatar.tsx
import React from "react";

type AvatarProps = {
  src?: string | null;
  alt?: string;
  size?: number;
  className?: string;
  fallbackText?: string; // e.g., initials
};

function svgFallbackDataURI(label: string) {
  const safe = (label || "PS").slice(0, 2).toUpperCase();
  const svg = `
  <svg xmlns="http://www.w3.org/2000/svg" width="128" height="128">
    <defs>
      <linearGradient id="g" x1="0" y1="0" x2="1" y2="1">
        <stop offset="0" stop-color="#1EF2FF" stop-opacity="0.28"/>
        <stop offset="1" stop-color="#00C7C2" stop-opacity="0.22"/>
      </linearGradient>
    </defs>
    <rect width="128" height="128" rx="28" fill="#0C111A"/>
    <rect width="128" height="128" rx="28" fill="url(#g)"/>
    <text x="50%" y="54%" text-anchor="middle" dominant-baseline="middle"
      font-family="system-ui, -apple-system, Segoe UI, Roboto, Helvetica, Arial"
      font-size="44" font-weight="700" fill="#F5F7FA">${safe}</text>
  </svg>`;
  return `data:image/svg+xml;charset=utf-8,${encodeURIComponent(svg)}`;
}

export function Avatar({
  src,
  alt = "avatar",
  size = 32,
  className = "",
  fallbackText = "PS",
}: AvatarProps) {
  const [broken, setBroken] = React.useState(false);

  const finalSrc = !src || broken ? svgFallbackDataURI(fallbackText) : src;

  return (
    <img
      src={finalSrc}
      alt={alt}
      width={size}
      height={size}
      className={className}
      style={{
        width: size,
        height: size,
        borderRadius: Math.round(size * 0.35),
        objectFit: "cover",
      }}
      onError={() => setBroken(true)}
      loading="lazy"
      decoding="async"
    />
  );
}

// client/src/utils/safeJson.ts
export function safeJsonParse<T>(raw: string | null, fallback: T): T {
  if (!raw) return fallback;
  try {
    return JSON.parse(raw) as T;
  } catch {
    return fallback;
  }
}

export function safeJsonStringify(value: unknown): string | null {
  try {
    return JSON.stringify(value);
  } catch {
    return null;
  }
}

// client/src/context/FullscreenContext.tsx
import React from "react";

type FullscreenContextValue = {
  isFullscreen: boolean;
  isSupported: boolean;
  enter: () => Promise<void>;
  exit: () => Promise<void>;
  toggle: () => Promise<void>;
};

const FullscreenContext = React.createContext<FullscreenContextValue | null>(null);

function getFullscreenElement(): Element | null {
  return document.fullscreenElement ?? null;
}

async function requestFullscreen(el: HTMLElement) {
  // Standard API
  if (el.requestFullscreen) return el.requestFullscreen();
  // If some older webviews exist, fail safely
  return Promise.reject(new Error("Fullscreen request not supported"));
}

async function exitFullscreen() {
  if (document.exitFullscreen) return document.exitFullscreen();
  return Promise.reject(new Error("Fullscreen exit not supported"));
}

export function FullscreenProvider({ children }: { children: React.ReactNode }) {
  const [isFullscreen, setIsFullscreen] = React.useState(false);

  const isSupported =
    typeof document !== "undefined" &&
    typeof document.fullscreenEnabled !== "undefined" &&
    document.fullscreenEnabled === true;

  React.useEffect(() => {
    const onChange = () => setIsFullscreen(!!getFullscreenElement());
    document.addEventListener("fullscreenchange", onChange);
    onChange();
    return () => document.removeEventListener("fullscreenchange", onChange);
  }, []);

  const enter = React.useCallback(async () => {
    if (!isSupported) return;
    const target = document.documentElement as unknown as HTMLElement;
    try {
      await requestFullscreen(target);
    } catch {
      // swallow: reported issue is “Fullscreen API errors” (avoid crashing / noisy console loops)
    }
  }, [isSupported]);

  const exit = React.useCallback(async () => {
    if (!isSupported) return;
    try {
      if (getFullscreenElement()) await exitFullscreen();
    } catch {
      // swallow
    }
  }, [isSupported]);

  const toggle = React.useCallback(async () => {
    if (!isSupported) return;
    if (getFullscreenElement()) return exit();
    return enter();
  }, [enter, exit, isSupported]);

  const value: FullscreenContextValue = { isFullscreen, isSupported, enter, exit, toggle };
  return <FullscreenContext.Provider value={value}>{children}</FullscreenContext.Provider>;
}

export function useFullscreen() {
  const ctx = React.useContext(FullscreenContext);
  if (!ctx) throw new Error("useFullscreen must be used within FullscreenProvider");
  return ctx;
}

// client/src/features/ai/AIAssistant.storage.ts
import { safeJsonParse, safeJsonStringify } from "@/utils/safeJson";

export type ChatMessage = {
  id: string;
  role: "user" | "assistant" | "system";
  content: string;
  createdAt: number;
};

const KEY = "perthsaver.ai.chatHistory.v1";
const MAX = 80;

export function loadChatHistory(): ChatMessage[] {
  const arr = safeJsonParse<unknown>(localStorage.getItem(KEY), []);
  if (!Array.isArray(arr)) return [];
  return arr
    .filter((m) => m && typeof m === "object")
    .map((m: any) => ({
      id: typeof m.id === "string" ? m.id : crypto.randomUUID(),
      role: m.role === "user" || m.role === "assistant" || m.role === "system" ? m.role : "assistant",
      content: typeof m.content === "string" ? m.content : "",
      createdAt: typeof m.createdAt === "number" ? m.createdAt : Date.now(),
    }))
    .filter((m) => m.content.trim().length > 0)
    .slice(-MAX);
}

export function saveChatHistory(messages: ChatMessage[]) {
  const bounded = (messages ?? []).slice(-MAX);
  const payload = safeJsonStringify(bounded);
  if (!payload) return;
  try {
    localStorage.setItem(KEY, payload);
  } catch {
    // storage full / denied — fail safely
  }
}

import React from "react";
import { loadChatHistory, saveChatHistory, ChatMessage } from "./AIAssistant.storage";

export function AIAssistant() {
  const [messages, setMessages] = React.useState<ChatMessage[]>([]);
  const [error, setError] = React.useState<string | null>(null);

  React.useEffect(() => {
    try {
      setMessages(loadChatHistory());
    } catch {
      setMessages([]);
    }
  }, []);

  React.useEffect(() => {
    try {
      saveChatHistory(messages);
    } catch {
      // no-op
    }
  }, [messages]);

  async function send(prompt: string) {
    setError(null);

    const userMsg: ChatMessage = {
      id: crypto.randomUUID(),
      role: "user",
      content: prompt,
      createdAt: Date.now(),
    };

    setMessages((prev) => [...prev, userMsg]);

    try {
      const res = await fetch("/api/ai/chat", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ messages: [...messages, userMsg] }),
      });

      if (!res.ok) {
        const txt = await res.text().catch(() => "");
        throw new Error(txt || `AI request failed (${res.status})`);
      }

      const data = (await res.json()) as { reply?: string };
      const reply = typeof data.reply === "string" ? data.reply : "";

      if (!reply.trim()) throw new Error("AI returned an empty response.");

      const aiMsg: ChatMessage = {
        id: crypto.randomUUID(),
        role: "assistant",
        content: reply,
        createdAt: Date.now(),
      };

      setMessages((prev) => [...prev, aiMsg]);
    } catch (e: any) {
      setError(e?.message || "AI error");
      // Optional: append a visible assistant error bubble rather than crashing
      setMessages((prev) => [
        ...prev,
        {
          id: crypto.randomUUID(),
          role: "assistant",
          content: "Sorry — I hit an error processing that. Please try again.",
          createdAt: Date.now(),
        },
      ]);
    }
  }

  return (
    <div>
      {error ? <div style={{ color: "#FF4D4D", fontSize: 12 }}>{error}</div> : null}
      {/* keep your existing UI; just wire it to send() */}
    </div>
  );
}

// client/src/billing/stripeParsing.ts
type StripePrice = {
  currency?: string;
  unit_amount?: number | null;
  unit_amount_decimal?: string | null;
  recurring?: { interval?: string | null } | null;
};

export function parseStripeUnitAmountCents(price: StripePrice): number {
  // Stripe can return:
  // - unit_amount: number | null
  // - unit_amount_decimal: string | null (e.g. "999" for $9.99 if currency uses 2 decimals)
  if (typeof price.unit_amount === "number" && Number.isFinite(price.unit_amount)) return price.unit_amount;

  const dec = price.unit_amount_decimal;
  if (typeof dec === "string" && dec.trim() !== "") {
    const n = Number(dec);
    if (Number.isFinite(n)) return Math.round(n);
  }

  return 0;
}

export function formatCurrencyFromCents(cents: number, currency: string = "aud"): string {
  const safe = Number.isFinite(cents) ? cents : 0;
  const amount = safe / 100;
  try {
    return new Intl.NumberFormat("en-AU", {
      style: "currency",
      currency: currency.toUpperCase(),
      maximumFractionDigits: 2,
    }).format(amount);
  } catch {
    return `$${amount.toFixed(2)}`;
  }
}
